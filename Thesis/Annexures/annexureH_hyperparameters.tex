\subsection{Bayesian Optimisation: GP Surrogate Methodology}
\label{annexure:bo_methodology}

\begin{lstlisting}[caption={GP surrogate surface fitting function.}]
from sklearn.gaussian_process import (
    GaussianProcessRegressor)
from sklearn.gaussian_process.kernels import (
    Matern, ConstantKernel, WhiteKernel)

def fit_gp_surface(X_obs, y_obs, log_axes,
                   grid_resolution=100):
    """Fit a GP surrogate to observed evaluations.

    Parameters
    ----------
    X_obs : np.ndarray, shape (n, 2)
        Observed hyperparameter configurations.
    y_obs : np.ndarray, shape (n,)
        Observed objective (validation error).
    log_axes : list of bool, length 2
        Whether to transform each axis to log10.
    grid_resolution : int
        Grid points per axis for surface eval.

    Returns
    -------
    gp, X1d, X2d, Zm, Zs, min_idx
    """
    X_gp = X_obs.copy()
    for i, la in enumerate(log_axes):
        if la:
            X_gp[:, i] = np.log10(X_gp[:, i])

    kernel = (
        ConstantKernel(1.0, (1e-4, 1e4)) *
        Matern(length_scale=[1.0, 1.0], nu=2.5,
               length_scale_bounds=(1e-4, 1e4)) +
        WhiteKernel(noise_level=1e-6,
                    noise_level_bounds=(1e-10, 1e-2)))

    gp = GaussianProcessRegressor(
        kernel=kernel, n_restarts_optimizer=15,
        random_state=SEED, alpha=1e-8)
    gp.fit(X_gp, y_obs)

    # Evaluation grid with 15% margin
    bounds = []
    for i in range(2):
        c = X_gp[:, i]
        margin = (c.max() - c.min()) * 0.15
        bounds.append((c.min() - margin,
                       c.max() + margin))

    x1_g = np.linspace(bounds[0][0], bounds[0][1],
                       grid_resolution)
    x2_g = np.linspace(bounds[1][0], bounds[1][1],
                       grid_resolution)
    X1m, X2m = np.meshgrid(x1_g, x2_g)
    X_grid = np.column_stack(
        [X1m.ravel(), X2m.ravel()])

    Zm, Zs = gp.predict(X_grid, return_std=True)
    Zm = Zm.reshape(X1m.shape)
    Zs = Zs.reshape(X1m.shape)

    # Back-transform to display scale
    X1d = 10**X1m if log_axes[0] else X1m
    X2d = 10**X2m if log_axes[1] else X2m

    mi = np.unravel_index(Zm.argmin(), Zm.shape)

    return gp, X1d, X2d, Zm, Zs, mi
\end{lstlisting}
