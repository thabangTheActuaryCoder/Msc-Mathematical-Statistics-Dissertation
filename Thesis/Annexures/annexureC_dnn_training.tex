\section{DNN Architecture and Training Procedure}
\label{annexure:dnn_training}

The FunnelDNN model uses a DNN architecture with monotonically decreasing hidden layer widths: $22 \to 256 \to 128 \to 64 \to 32 \to 16 \to 1$. All layers are bias-free, and the CELU activation function is applied after each hidden layer. The output layer is linear with no activation.

% -------------------------------------------------------------
\subsection{Training Procedure}
\label{annexure:train_model}

Training minimises the MSE loss on normalised data using the NAdam optimiser. Validation-based early stopping monitors the dot-product \ensuremath{R^2} on the validation set every 5 epochs, with a patience of 10 checks.

\begin{algorithm}[htbp]
\caption{FunnelDNN training with early stopping.}
\label{alg:dnn_training}
\begin{algorithmic}[1]
\Require Standardised data $(\mathbf{X}_{\mathrm{train}}, \mathbf{y}_{\mathrm{train}}, \mathbf{X}_{\mathrm{val}}, \mathbf{y}_{\mathrm{val}})$, learning rate $\eta$, batch size $B$, patience $P$, validation interval $V$
\Ensure Trained model state $\boldsymbol{\theta}^*$
\State Initialise FunnelDNN with seed 42
\State $\boldsymbol{\theta}^* \gets \boldsymbol{\theta}_0$, \; $R^{2*} \gets -\infty$, \; $c \gets 0$
\For{$t = 1, 2, \ldots, T_{\max}$}
    \For{each mini-batch $(\mathbf{X}_b, \mathbf{y}_b)$ of size $B$}
        \State Compute $\mathcal{L} = \mathrm{MSE}\bigl(f_{\boldsymbol{\theta}}(\mathbf{X}_b),\, \mathbf{y}_b\bigr)$
        \State Update $\boldsymbol{\theta}$ via NAdam with learning rate $\eta$
    \EndFor
    \If{$t \bmod V = 0$}
        \State Compute $R^2_{\mathrm{val}} = r^2_{\mathrm{dot}}\bigl(\mathbf{y}_{\mathrm{val}},\, f_{\boldsymbol{\theta}}(\mathbf{X}_{\mathrm{val}})\bigr)$
        \If{$R^2_{\mathrm{val}} > R^{2*}$}
            \State $R^{2*} \gets R^2_{\mathrm{val}}$, \; $\boldsymbol{\theta}^* \gets \boldsymbol{\theta}$, \; $c \gets 0$
        \Else
            \State $c \gets c + 1$
        \EndIf
        \If{$c \geq P$}
            \State \textbf{break} \Comment{Early stopping triggered}
        \EndIf
    \EndIf
\EndFor
\State \Return $\boldsymbol{\theta}^*$
\end{algorithmic}
\end{algorithm}

% -------------------------------------------------------------
\subsection{Evaluation Metrics}
\label{annexure:eval_helpers}

Three metrics are computed on the original Rand scale after de-normalisation:
\begin{itemize}
    \item $R^2 = 1 - \mathrm{SS}_{\mathrm{res}} / \mathrm{SS}_{\mathrm{tot}}$, where $\mathrm{SS}_{\mathrm{res}} = \sum_i (y_i - \hat{y}_i)^2$ and $\mathrm{SS}_{\mathrm{tot}} = \sum_i (y_i - \bar{y})^2$.
    \item $\mathrm{RMSE} = \sqrt{n^{-1} \sum_i (y_i - \hat{y}_i)^2}$.
    \item $\mathrm{MAE} = n^{-1} \sum_i |y_i - \hat{y}_i|$.
\end{itemize}

During training, the dot-product \ensuremath{R^2} (squared correlation between predicted and observed values) is used on normalised data for computational speed.

% -------------------------------------------------------------
\subsection{Checkpoint Contents}
\label{annexure:final_model}

The final trained model is saved as \texttt{chapter3\_final\_\allowbreak{}model.pth}, a PyTorch checkpoint containing:
\begin{itemize}
    \item Model weight tensors (state dictionary).
    \item Architecture specification (layer widths, activation, bias setting).
    \item Selected hyperparameters (learning rate, batch size, optimiser, momentum).
    \item Standardisation parameters ($\boldsymbol{\mu}_X$, $\boldsymbol{\sigma}_X$, $\mu_y$, $\sigma_y$).
    \item Feature names and evaluation metrics.
\end{itemize}

\subsection{DNN Training Algorithm (Detailed)}
\label{annexure:dnn_algorithm_detailed}

The following table presents the complete DNN training and model
selection procedure as implemented in
Section~\ref{sec:dnn_specification_training}.

\begin{table}[H]
\centering
{\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{p{0.92\textwidth}}
\toprule
\textbf{Algorithm C.1: DNN training and model selection} \\
\midrule
\textbf{Input:} model-ready training data
$\{(\mathbf{z}_i, \widetilde{y}_i)\}_{i \in \mathcal{D}_{\mathrm{train}}}$,
model-ready validation data
$\{(\mathbf{z}_i, \widetilde{y}_i)\}_{i \in \mathcal{D}_{\mathrm{val}}}$,
architecture specification $(d_0, d_1, \ldots, d_L)$,
activation function $g(\cdot)$, loss function $\mathcal{L}_{\mathrm{MSE}}$,
optimiser, learning rate $\eta$, momentum parameters $(\beta_1,\beta_2)$,
batch size $|\mathcal{B}|$,
maximum epochs $E_{\max} = 200$, validation frequency $v = 5$,
early stopping patience $P = 10$, random seed $s = 42$. \\[3pt]

\textbf{Output:} fitted network weights $\widehat{\theta}$,
training history, selected epoch $e^{*}$. \\[3pt]

1. Set the random seed to $s$. \\
2. Initialise network weights $\theta^{(0)}$ using Kaiming uniform
   initialisation \citep{he2015delving}. \\
3. For epoch $e = 1, \ldots, E_{\max}$: \\
\quad (a) Partition $\mathcal{D}_{\mathrm{train}}$ into random
        mini-batches of size $|\mathcal{B}|$. \\
\quad (b) For each mini-batch $\mathcal{B}$, compute
        $\mathcal{L}_{\mathrm{MSE}}(\theta; \mathcal{B})$ and update
        $\theta$ using the optimiser. \\
\quad (c) Record training loss $\mathcal{L}_{\mathrm{train}}(e)$. \\
\quad (d) If $e \bmod v = 0$: evaluate validation loss
        $\mathcal{L}_{\mathrm{val}}(e)$ and dot-product $R^2$ on both
        partitions.  If validation $R^2$ improves, checkpoint the model
        weights; otherwise, increment the patience counter. \\
\quad (e) If the patience counter reaches $P$, terminate training. \\
4. Select $e^{*} = \arg\max_{e \in \mathcal{E}_{\mathrm{val}}}
   R^2_{\mathrm{val}}(e)$. \\
5. Retain the network weights $\widehat{\theta} = \theta^{(e^{*})}$. \\
6. De-normalise predictions and evaluate on the held-out test set using
   \ensuremath{R^2}, RMSE and MAE on the original monetary scale. \\
7. Save $\widehat{\theta}$, architecture specification, standardisation
   parameters, training history, and all evaluation outputs. \\
\bottomrule
\end{tabular}}
\caption{DNN training and model selection algorithm.}
\label{tab:dnn_training_algorithm_annex}
\end{table}

\subsection{GLM Fitting Algorithm}
\label{annexure:glm_algorithm}

\begin{table}[H]
\centering
{\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{p{0.92\textwidth}}
\toprule
\textbf{Algorithm C.2: Gamma GLM with log link} \\
\midrule
\textbf{Input:} model-ready training data
$\{(\mathbf{z}_i, y_i)\}_{i \in \mathcal{D}_{\mathrm{train}}}$,
distribution family (gamma), link function ($\log$). \\[3pt]

\textbf{Output:} fitted coefficient vector
$\widehat{\boldsymbol{\beta}}$, dispersion estimate
$\widehat{\phi}$. \\[3pt]

1. Initialise the linear predictor:
   $\eta_i^{(0)} = \log(\bar{y}_{\mathrm{train}})$ for all $i$. \\
2. Repeat until convergence: \\
\quad (a) Compute fitted values:
        $\mu_i^{(t)} = \exp(\eta_i^{(t)})$. \\
\quad (b) Compute working weights:
        $w_i^{(t)} = \mu_i^{(t)\,2} / (\phi\,\mu_i^{(t)\,2})
        = 1/\phi$. \\
\quad (c) Compute the adjusted dependent variable:
        $\tilde{y}_i^{(t)} = \eta_i^{(t)}
        + (y_i - \mu_i^{(t)}) / \mu_i^{(t)}$. \\
\quad (d) Solve the weighted least squares problem:
        $\boldsymbol{\beta}^{(t+1)}
        = (\mathbf{Z}^{\top} \mathbf{W}^{(t)} \mathbf{Z})^{-1}
          \mathbf{Z}^{\top} \mathbf{W}^{(t)}
          \tilde{\mathbf{y}}^{(t)}$. \\
\quad (e) Update the linear predictor:
        $\eta_i^{(t+1)} = \mathbf{z}_i^{\top}
        \boldsymbol{\beta}^{(t+1)}$. \\
3. Estimate the dispersion parameter:
   $\widehat{\phi} = \frac{1}{n - p - 1}
   \sum_{i=1}^{n}
   \bigl((y_i - \widehat{\mu}_i)/\widehat{\mu}_i\bigr)^{2}$. \\
4. Predict on the original monetary scale:
   $\widehat{y}_i = \exp(\mathbf{z}_i^{\top}
   \widehat{\boldsymbol{\beta}})$. \\
5. Evaluate \ensuremath{R^2}, RMSE and MAE on the training,
   validation and test partitions. \\
\bottomrule
\end{tabular}}
\caption{Gamma GLM with log link fitting algorithm.}
\label{tab:glm_algorithm_annex}
\end{table}

\subsection{RF Training Algorithm}
\label{annexure:rf_algorithm}

\begin{table}[H]
\centering
{\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{p{0.92\textwidth}}
\toprule
\textbf{Algorithm C.3: RF regression} \\
\midrule
\textbf{Input:} model-ready training data
$\{(\mathbf{z}_i, y_i)\}_{i \in \mathcal{D}_{\mathrm{train}}}$,
number of trees $B = 500$, number of candidate predictors per split
$m = \lfloor p/3 \rfloor$, minimum samples per leaf
$n_{\min} = 5$, random seed $s = 42$. \\[3pt]

\textbf{Output:} ensemble of fitted trees
$\{T_1, \ldots, T_B\}$. \\[3pt]

1. Set the random seed to $s$. \\
2. For $b = 1, \ldots, B$: \\
\quad (a) Draw a bootstrap sample $\mathcal{D}_{b}^{*}$ of size
        $n_{\mathrm{train}}$ with replacement from
        $\mathcal{D}_{\mathrm{train}}$. \\
\quad (b) Grow an unpruned regression tree $T_b$ on
        $\mathcal{D}_{b}^{*}$: \\
\quad\quad (i) At each node, select a random subset
            $\mathcal{S} \subset \{1,\ldots,p\}$ with
            $|\mathcal{S}| = m$. \\
\quad\quad (ii) Choose the split variable $j^{*}$ and split point
            $s^{*}$ from $\mathcal{S}$ that minimise the weighted
            residual sum of squares. \\
\quad\quad (iii) Split the node into two child nodes and repeat
            until each terminal node contains at most $n_{\min}$
            observations. \\
\quad (c) Assign the sample mean of the response values in each
        terminal node as the node prediction. \\
3. For a new input $\mathbf{z}$, predict
   $\hat{f}_{\mathrm{RF}}(\mathbf{z})
   = \frac{1}{B}\sum_{b=1}^{B} T_b(\mathbf{z})$. \\
4. Evaluate \ensuremath{R^2}, RMSE and MAE on the training,
   validation and test partitions on the original monetary scale. \\
\bottomrule
\end{tabular}}
\caption{RF regression training algorithm.}
\label{tab:rf_algorithm_annex}
\end{table}

\subsection{XGBoost Training Algorithm}
\label{annexure:xgb_algorithm}

\begin{table}[H]
\centering
{\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{p{0.92\textwidth}}
\toprule
\textbf{Algorithm C.4: XGBoost gradient-boosted tree regression} \\
\midrule
\textbf{Input:} model-ready training data
$\{(\mathbf{z}_i, y_i)\}_{i \in \mathcal{D}_{\mathrm{train}}}$,
model-ready validation data
$\{(\mathbf{z}_i, y_i)\}_{i \in \mathcal{D}_{\mathrm{val}}}$,
maximum boosting rounds $M = 1\,000$, learning rate $\eta = 0.05$,
maximum tree depth $d_{\max} = 8$, $\mathcal{L}_2$ regularisation
$\lambda$, minimum split gain $\gamma$, early stopping patience
$P = 50$ rounds, random seed $s = 42$. \\[3pt]

\textbf{Output:} fitted additive ensemble
$\hat{f}_{M^{*}}(\mathbf{z})$, selected round $M^{*}$. \\[3pt]

1. Set the random seed to $s$. \\
2. Initialise the prediction:
   $\hat{f}_{0}(\mathbf{z}_i) = \bar{y}_{\mathrm{train}}$
   for all $i$. \\
3. For boosting round $m = 1, \ldots, M$: \\
\quad (a) Compute gradient statistics $g_i$ and $h_i$ for each
        training observation. \\
\quad (b) Grow a regression tree $h_m$ of maximum depth $d_{\max}$: \\
\quad\quad (i) At each node, evaluate all candidate split points
            across all $p$ predictors. \\
\quad\quad (ii) Select the split $(j^{*}, s^{*})$ that maximises
            the gain. \\
\quad\quad (iii) Split only if $\mathrm{Gain} > 0$; otherwise the
            node becomes a leaf. \\
\quad\quad (iv) Assign optimal leaf weights
            $w_j^{*} = -G_j / (H_j + \lambda)$. \\
\quad (c) Update predictions:
        $\hat{f}_{m}(\mathbf{z}_i)
        = \hat{f}_{m-1}(\mathbf{z}_i) + \eta \, h_m(\mathbf{z}_i)$. \\
\quad (d) Evaluate validation loss
        $\mathcal{L}_{\mathrm{val}}(m)$. If the validation loss
        has not improved for $P$ consecutive rounds, terminate
        training. \\
4. Select the round
   $M^{*} = \arg\min_{m \in \{1,\ldots,M\}}
   \mathcal{L}_{\mathrm{val}}(m)$
   and retain the ensemble up to that round. \\
5. Evaluate \ensuremath{R^2}, RMSE and MAE on the training,
   validation and test partitions on the original monetary scale. \\
\bottomrule
\end{tabular}}
\caption{XGBoost gradient-boosted tree regression training algorithm.}
\label{tab:xgb_algorithm_annex}
\end{table}
